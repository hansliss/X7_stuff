cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
        set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_LIST_DIR}/../cmake/x7-mips-toolchain.cmake" CACHE FILEPATH "X7 toolchain")
    endif()
    project(omega_x7 LANGUAGES C ASM)
    include(../cmake/X7Probe.cmake)
endif()
set(APP_BASE "0x69800000" CACHE STRING "Application load base")
option(X7_OMEGA_DEBUG_LOG "Write routine Omega Race startup/performance logs" OFF)
add_executable(omega-race vendor/fastz80/z80.c machine.c render.c platform.c nvram.c runtime.c
    ../doom_x7/input_stubs.S ../sysprobe/crt.c ../sysprobe/app_stubs.S
    ../sysprobe/ui_stubs.S ../sysprobe/audio_stubs.S ../sysprobe/fb_video.c)
set_target_properties(omega-race PROPERTIES SUFFIX ".app")
set(HEADERS "${CMAKE_CURRENT_BINARY_DIR}/headers")
file(MAKE_DIRECTORY "${HEADERS}")
file(WRITE "${HEADERS}/string.h" "#include <stddef.h>\nvoid *memset(void *, int, size_t);\nvoid *memcpy(void *, const void *, size_t);\n")
target_include_directories(omega-race PRIVATE "${HEADERS}")
target_compile_options(omega-race PRIVATE -fwrapv -fno-strict-aliasing)
target_compile_definitions(omega-race PRIVATE X7_OMEGA_DEBUG_LOG=$<BOOL:${X7_OMEGA_DEBUG_LOG}>)
x7_probe_target(omega-race "${CMAKE_CURRENT_SOURCE_DIR}/../sysprobe/app.xn")
target_link_options(omega-race PRIVATE "-Wl,--defsym=APP_BASE=${APP_BASE}")
add_custom_command(TARGET omega-race POST_BUILD
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../sysprobe/check_app.py" "$<TARGET_FILE:omega-race>" VERBATIM)
